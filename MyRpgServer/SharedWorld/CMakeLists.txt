cmake_minimum_required(VERSION 3.20)
project(SharedWorldCore LANGUAGES CXX)
find_package(Threads REQUIRED)
add_library(SharedWorldCore INTERFACE)
target_compile_features(SharedWorldCore INTERFACE cxx_std_17)
target_include_directories(SharedWorldCore INTERFACE "${CMAKE_CURRENT_SOURCE_DIR}/..")
target_link_libraries(SharedWorldCore INTERFACE Threads::Threads)
add_executable(WorldCoreTests ../Tests/WorldCoreTests.cpp)
target_link_libraries(WorldCoreTests PRIVATE SharedWorldCore)
if(MSVC)
    target_compile_options(WorldCoreTests PRIVATE /W4 /WX /permissive-)
else()
    target_compile_options(WorldCoreTests PRIVATE -Wall -Wextra -Werror -pedantic)
endif()
enable_testing()
add_test(NAME WorldCoreTests COMMAND WorldCoreTests)
add_executable(MapBakeTool Tools/MapBakeTool/main.cpp)
target_link_libraries(MapBakeTool PRIVATE SharedWorldCore)
add_executable(WorldMapInspect Tools/MapBakeTool/inspect.cpp)
target_link_libraries(WorldMapInspect PRIVATE SharedWorldCore)
add_executable(WorldMovementInspect Tools/MapBakeTool/movement_inspect.cpp)
target_link_libraries(WorldMovementInspect PRIVATE SharedWorldCore)
add_executable(WorldChunkBake Tools/MapBakeTool/chunks_main.cpp)
target_link_libraries(WorldChunkBake PRIVATE SharedWorldCore)
add_executable(WorldChunkWalk Tools/MapBakeTool/chunk_walk.cpp)
target_link_libraries(WorldChunkWalk PRIVATE SharedWorldCore)
